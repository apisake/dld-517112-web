12.1.2Verilog กับ VHDL: half adder

module half_adder (
  input  a, b,
  output s, c
);
  assign s = a ^ b;
  assign c = a & b;
endmodule
library ieee;
use ieee.std_logic_1164.all;
entity half_adder is
  port (a, b : in  std_logic;
        s, c : out std_logic);
end half_adder;
architecture dataflow of half_adder is
begin
  s <= a xor b;
  c <= a and b;
end dataflow;

Verilog กระชับคล้ายภาษา C · VHDL ละเอียดและเข้มงวดเรื่องชนิดข้อมูล — บทนี้ใช้ Verilog

12.2โครงสร้างของโมดูล

module ชื่อโมดูล (
  input  ชื่ออินพุต, ...,
  output ชื่อเอาต์พุต, ...
);
  // ประกาศสัญญาณภายใน (wire / reg)
  // assign / always / โมดูลย่อย
endmodule
wire [3:0] a;     // a[3] = MSB
reg  [7:0] count;
assign x = a[2];    // bit-select
assign y = a[3:2];  // part-select

ค่าคงที่: ขนาด'ฐาน ค่า
4'b1010 · 8'hA5 · 4'd9 · 1'b0

wire = สายเชื่อม (assign) · reg = กำหนดค่าใน always

12.2.4ลองเอง: ตัวดำเนินการกับเวกเตอร์

คลิกบิตของ a, b แล้วดูทุกเอาต์พุต · ถาม: ทำไม sum ต้อง 5 บิต · &a คือ reduction · แก้โค้ดสดได้ เช่น เพิ่ม assign y = a[3:2];

12.3.1แบบโครงสร้าง: gate primitive

// full adder (เทียบรูปที่ 7.5)
module fa_struct (
  input  a, b, cin,
  output s, cout
);
  wire p, g, t;
  xor u1 (p, a, b);
  xor u2 (s, p, cin);
  and u3 (g, a, b);
  and u4 (t, p, cin);
  or  u5 (cout, g, t);
endmodule
รูปที่ 12.2 · พอร์ตแรกของ primitive เป็นเอาต์พุต

12.3.2แบบ dataflow: assign

module fa_dataflow (
  input  a, b, cin,
  output s, cout
);
  assign s    = a ^ b ^ cin;
  assign cout = (a & b) | (cin & (a ^ b));
endmodule
// 2-to-1 MUX ขนาด 4 บิต
module mux2 (
  input  [3:0] d0, d1,
  input        s,
  output [3:0] y
);
  assign y = s ? d1 : d0;
endmodule

นิพจน์เดียวกับหัวข้อ 7.1.2 · ทุก assign ทำงานพร้อมกันตลอดเวลา — ลำดับบรรทัดไม่มีผล

ลองเอง: แก้โค้ดแล้วดูตารางค่าความจริง

ลองแก้ cout เป็น (a & b) | (b & cin) | (a & cin) — ตารางเหมือนเดิม · เลือก gray4 แล้วอธิบาย b ^ (b >> 1)

12.3.3–12.3.4แบบพฤติกรรม และลำดับชั้น

module mux4 (
  input      [3:0] d,
  input      [1:0] s,
  output reg       y
);
  always @* begin
    case (s)
      2'b00: y = d[0];
      2'b01: y = d[1];
      2'b10: y = d[2];
      default: y = d[3];
    endcase
  end
endmodule
รูปที่ 12.3 adder4 = fa_struct × 4 (โปรแกรมที่ 12.9)
fa_struct fa0 (.a(a[0]), .b(b[0]), .cin(cin),
               .s(s[0]), .cout(c1));
always @* = ทำงานทุกครั้งที่สัญญาณใดใน always เปลี่ยน → วงจรคอมไบเนชัน · เอาต์พุตต้องเป็น reg · การต่อแบบลำดับชั้นเหมือนต่อไอซี 74 บนบอร์ด

12.3.5ระวัง latch ที่เกิดโดยไม่ตั้งใจ

กฎ 3 ข้อ: ทุก if มี else · ทุก case มี default · หรือกำหนดค่าตั้งต้นที่บรรทัดแรกของ always @*

12.4.1วงจรเชิงลำดับ: always @(posedge clk)

module dff_ar (
  input      clk, rst_n, d,
  output reg q
);
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n)
      q <= 1'b0;     // reset ทันที
    else
      q <= d;        // รับ d ที่ขอบขาขึ้น
  end
endmodule
  • posedge clk → D flip-flop ขอบขาขึ้น (บทที่ 8)
  • negedge rst_n ในรายการ → reset asynchronous active-LOW (เทียบ CLR)
  • ใช้ <= (non-blocking) เสมอในวงจรเชิงลำดับ

12.4.2Non-blocking (<=) กับ blocking (=)

กดขอบนาฬิกาหลายครั้งโดยสลับ d — nb q2 ช้ากว่า q1 หนึ่งคาบ แต่ b q2 = q1 เสมอ (รูปที่ 12.4, ตารางที่ 12.3) · swap ทำได้เฉพาะ non-blocking

12.4.3วงจรนับขึ้น/ลง 4 บิต

นับขึ้นจนเกิน 15 → 0 · ตั้ง up = 0 · ตั้ง en = 0 แล้วดูบรรทัดที่ทำงาน · กด rst_n = 0 แล้วดูว่า reset ทันที (ผังเวลารูปที่ 12.9)

12.4.4–12.4.5ชิฟต์รีจิสเตอร์ และ state machine

always @(posedge clk or negedge rst_n)
  if (!rst_n) q <= 4'b0000;
  else        q <= {din, q[3:1]};

รูปที่ 12.6 always สองบล็อก = รีจิสเตอร์สถานะ + วงจรคอมไบเนชัน

12.5Testbench แบบตรวจสอบตัวเอง

module tb_fa;                  // ไม่มีพอร์ต
  reg  a, b, cin;  wire s, cout;
  integer i, errors = 0;
  fa_struct dut (.a(a), .b(b), .cin(cin),
                 .s(s), .cout(cout));
  initial begin
    for (i = 0; i < 8; i = i + 1) begin
      {a, b, cin} = i;  #10;
      if ({cout, s} !== a + b + cin) begin
        $display("ERROR at a=%b b=%b cin=%b", a, b, cin);
        errors = errors + 1;
      end
    end
    if (errors == 0) $display("PASS");
    $finish;
  end
endmodule

ลองเอง: รัน testbench หาบั๊ก

รัน fa_struct → PASS · เลือก “มีบั๊ก” → ERROR ที่ a=0 b=1 cin=1 · ให้นักศึกษาหาบรรทัดที่ผิดจากกรณีที่ fail

12.6.4โค้ดที่สังเคราะห์ได้และไม่ได้

สังเคราะห์ได้ใช้ได้เฉพาะการจำลอง
assign, gate primitive, ตัวดำเนินการ#10 (หน่วงเวลา)
always @* ที่กำหนดค่าครบทุกกรณี$display, $finish, $monitor
always @(posedge clk) กับ <=initial (ใน testbench)
case / if / โมดูลย่อยลูปที่จำนวนรอบไม่คงที่
ผลจำลองถูกต้อง ≠ สังเคราะห์ได้ — ตรวจรายงานการสังเคราะห์ทุกครั้ง (latch, สัญญาณที่ไม่ได้ใช้)

สรุปท้ายบท

  • HDL อธิบายฮาร์ดแวร์ที่ทำงานพร้อมกัน · โค้ดเดียวใช้ทั้งจำลองและสังเคราะห์
  • โมดูล = พอร์ต + สัญญาณ (wire/reg, เวกเตอร์) + การอธิบายวงจร
  • คอมไบเนชัน: gate primitive · assign · always @* (กำหนดค่าครบ ไม่เกิด latch)
  • เชิงลำดับ: always @(posedge clk) กับ <= · state machine แบบ always สองบล็อก
  • testbench แบบตรวจสอบตัวเอง: ป้อนทุกกรณีและเทียบค่าอัตโนมัติ
  • สังเคราะห์: โค้ด → netlist → แมปลง LUT/ฟลิปฟลอป → bitstream (บทที่ 11)
